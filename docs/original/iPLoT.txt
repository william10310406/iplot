def is_any_tk_active():
    # Checks if the default root window exists and is initialized
    return tkinter._default_root is not None

def get_default_root_title():
    try:
        if tkinter._default_root and tkinter._default_root.winfo_exists():
            return tkinter._default_root.title()
    except (_tkinter.TclError, AttributeError):
        pass
    return "未找到作用中的主視窗"

def cmdEXCUTE_iPLoT(filename):
    print(filename)
    subDIR=os.path.dirname(filename)
    myName=os.path.basename(subDIR)
    pass
    print(filename)
    if os.path.exists(filename):
       cmds='global '+myName+'HOMEDIR'
       print('*****',cmds)
       exec(cmds,globals())
       cmds=myName+'HOMEDIR="'+subDIR+'"'
       print(cmds)
       exec(cmds,globals())
       exec(open(filename,'r',encoding='utf-8').read(),globals())
    else:
       sub2DIR=sorted([x for x in os.listdir(subDIR) if os.path.isdir(os.path.join(subDIR,x))])
       NN=len(sub2DIR)
       if NN>0:
          DY=1./NN
          subDIRWIN=tkinter.Toplevel()
          subDIRWIN.title(myName)
          subDIRWIN.geometry('200x600+760+0')
          iii=0
          for w in sub2DIR:
              B=tkinter.Button(subDIRWIN)
              B.configure(text=w)
              filename=os.path.join(os.path.join(subDIR,w),w+'.txt')
              myName=w
              cmds='global '+myName+'HOMEDIR'
              print('*****',cmds)
              exec(cmds,globals())
              cmds=myName+'HOMEDIR="'+os.path.dirname(filename)+'"'
              print(cmds)
              exec(cmds,globals())
              B.configure(command=lambda a=filename:cmdEXCUTE_iPLoT(a))
              #B.configure(command=lambda a=filename:exec(open(a,'r',encoding='utf-8').read(),globals()))
              B.place(relx=0,rely=iii*DY,relwidth=1,relheight=DY)
              iii+=1

def DPanel(iPLoTHOMEDI,iPLoTWIN):

       subDIR=iPLoTHOMEDIR

       if NN>0:
          DY=0.9/NN
          iii=0
